// ============================================
// DASHICONS CONFLICT DEBUGGER
// ============================================

console.log('=== DASHICONS CONFLICT DEBUGGER ===\n');

// 1. Check all loaded stylesheets
console.log('1️⃣ LOADED STYLESHEETS:');
console.log('━'.repeat(50));
const stylesheets = document.querySelectorAll('link[rel="stylesheet"]');
stylesheets.forEach((link, index) => {
    const href = link.href;
    const isPlugin = href.includes('/plugins/');
    const isTheme = href.includes('/themes/');

    console.log(`${index + 1}. ${isPlugin ? '🔌 PLUGIN' : isTheme ? '🎨 THEME' : '📄 OTHER'}`);
    console.log(`   URL: ${href}`);
});
console.log('\n');

// 2. Find all CSS rules affecting dashicons
console.log('2️⃣ CSS RULES AFFECTING DASHICONS:');
console.log('━'.repeat(50));
const dashiconRules = [];
for (let sheet of document.styleSheets) {
    try {
        const rules = sheet.cssRules || sheet.rules;
        for (let rule of rules) {
            if (rule.cssText &&
                (rule.selectorText?.includes('dashicon') ||
                 rule.cssText.includes('dashicon') ||
                 rule.cssText.includes('\\f') ||
                 rule.selectorText?.includes('.product-type-bundle'))) {
                dashiconRules.push({
                    sheet: sheet.href || 'inline',
                    selector: rule.selectorText,
                    css: rule.cssText
                });
            }
        }
    } catch (e) {
        // CORS or other access issues
    }
}

if (dashiconRules.length === 0) {
    console.log('✅ No suspicious dashicon rules found');
} else {
    console.log(`⚠️ Found ${dashiconRules.length} rules affecting dashicons:`);
    dashiconRules.forEach((rule, index) => {
        console.log(`\n${index + 1}. FROM: ${rule.sheet}`);
        console.log(`   SELECTOR: ${rule.selector}`);
        console.log(`   RULE: ${rule.css}`);
    });
}
console.log('\n');

// 3. Check computed styles on menu icons
console.log('3️⃣ MENU ICONS COMPUTED STYLES:');
console.log('━'.repeat(50));
const menuIcons = document.querySelectorAll('#adminmenu .dashicons');
console.log(`Found ${menuIcons.length} menu icons`);

if (menuIcons.length > 0) {
    const firstIcon = menuIcons[0];
    const styles = window.getComputedStyle(firstIcon, ':before');
    console.log('\nFirst menu icon (:before pseudo-element):');
    console.log(`   content: ${styles.content}`);
    console.log(`   font-family: ${styles.fontFamily}`);
    console.log(`   Expected: "dashicons"`);

    if (styles.fontFamily !== 'dashicons' && !styles.fontFamily.includes('dashicons')) {
        console.log('   ⚠️ WARNING: Font-family is not dashicons!');
    }
}
console.log('\n');

// 4. Check body classes that might trigger rules
console.log('4️⃣ BODY CLASSES:');
console.log('━'.repeat(50));
const bodyClasses = document.body.className.split(' ');
console.log('Body classes that might affect styling:');
bodyClasses.forEach(cls => {
    if (cls.includes('bundle') || cls.includes('product')) {
        console.log(`   🔍 ${cls}`);
    }
});
console.log('\n');

// 5. Check for JavaScript that modifies icons
console.log('5️⃣ CHECKING FOR JS MODIFICATIONS:');
console.log('━'.repeat(50));
const observer = new MutationObserver((mutations) => {
    mutations.forEach((mutation) => {
        if (mutation.target.classList?.contains('dashicons')) {
            console.log('⚠️ DASHICON MODIFIED BY JAVASCRIPT!');
            console.log('   Element:', mutation.target);
            console.log('   Type:', mutation.type);
            console.log('   Stack trace:');
            console.trace();
        }
    });
});

// Observe for 5 seconds
const icons = document.querySelectorAll('.dashicons');
icons.forEach(icon => observer.observe(icon, {
    attributes: true,
    attributeOldValue: true
}));

console.log('Watching for icon changes for 5 seconds...');
setTimeout(() => {
    observer.disconnect();
    console.log('✅ Observation complete\n');
}, 5000);

// 6. Check active plugins
console.log('6️⃣ ACTIVE PLUGINS (from stylesheets):');
console.log('━'.repeat(50));
const plugins = new Set();
stylesheets.forEach(link => {
    const match = link.href.match(/\/plugins\/([^\/]+)\//);
    if (match) {
        plugins.add(match[1]);
    }
});
plugins.forEach(plugin => console.log(`   🔌 ${plugin}`));
console.log('\n');

// 7. Specific check for our plugin
console.log('7️⃣ ADVANCED BUNDLE SYSTEM CHECK:');
console.log('━'.repeat(50));
const absCSS = Array.from(stylesheets).find(s => s.href.includes('advanced-bundle-system'));
if (absCSS) {
    console.log('✅ Plugin CSS loaded:', absCSS.href);
} else {
    console.log('❌ Plugin CSS not found');
}

console.log('\n=== END OF REPORT ===');
console.log('📋 Copy all the output above and send it for analysis');

Alternative: Quick Plugin Test

To identify which plugin is causing the issue:

  1. Go to WordPress → Plugins
  2. Deactivate Advanced Bundle System
  3. Check if icons are still broken
  4. If YES: It's another plugin! Start deactivating others one by one
  5. If NO: It's our plugin, and we need to investigate more

Common Culprits:

  • 🔌 WooCommerce extensions that modify admin UI
  • 🎨 Admin theme/UI customization plugins
  • ⚡ Page builder plugins
  • 📊 Dashboard/analytics plugins
  • 🎯 Menu management plugins
